Q11. Install and configure Apache Cassandra. Create a keyspace named CollegeDB with an appropriate replication factor and create a Students table. Insert and retrieve at least five student records.

# Start Cassandra and open cqlsh
cqlsh

CREATE KEYSPACE CollegeDB
WITH replication = {'class':'SimpleStrategy','replication_factor':1};

USE CollegeDB;

CREATE TABLE Students (
  student_id int PRIMARY KEY,
  name text,
  course text,
  semester int,
  marks int
);
