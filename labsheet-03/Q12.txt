Q12. Using Apache Cassandra, perform CRUD operations on an Employees table. Demonstrate INSERT, SELECT, UPDATE, and DELETE operations using CQL.

USE CollegeDB;

CREATE TABLE Employees (
  employee_id int PRIMARY KEY,
  name text,
  department text,
  salary int
);

INSERT INTO Employees (employee_id,name,department,salary)
VALUES (101,'Aman','IT',30000);

SELECT * FROM Employees;

UPDATE Employees SET salary = 35000 WHERE employee_id = 101;

SELECT * FROM Employees WHERE employee_id = 101;

DELETE FROM Employees WHERE employee_id = 101;
